\ParallelTitle{Learning notes}{学习笔记}
\ParallelSection{sample-concept}{Average speed}{平均速率}
\ParallelText{sample-explanation}
 {Average speed is total distance divided by positive elapsed time. Include pauses in the interval. A trip of 180 metres in 12 seconds has average speed 15 metres per second.}
 {平均速率等于总路程除以大于零的总时间。计时区间应包含停顿。若行程为 180 米、用时 12 秒，平均速率就是每秒 15 米。}
\begin{ParallelKeep}
\ParallelText{sample-formula}
 {\[v_{\mathrm{avg}}=\frac{d}{\Delta t},\qquad \Delta t>0\]}
 {\[v_{\mathrm{avg}}=\frac{d}{\Delta t},\qquad \Delta t>0\]}
\ParallelText{sample-limit}
 {This average does not determine the speed at any particular instant.}
 {这一平均值不能确定任意特定时刻的速率。}
\end{ParallelKeep}
